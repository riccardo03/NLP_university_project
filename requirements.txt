#  PoliMillionaire — Python dependencies
#  Runtime: Python 3.10–3.12 | GPU: Colab T4 / A100 (CUDA 12.x)

#  ⚠ DO NOT install torch manually.
#  Google Colab ships the correct torch + torchvision + triton bundle for the
#  available CUDA version.  Installing a different torch wheel (e.g. cu121 on

# ── Numeric foundation ───────────────────────────────────────────────────────
# numpy 2.x is INCOMPATIBLE with faiss-cpu==1.8.0 and datasets==2.20.0.
# Keep strictly below 2.0.
numpy>=1.24.0,<2.0.0

# scipy is used inside the Maths RAG PAL sandbox (scipy.stats)
scipy>=1.11.0


# ── Hugging Face stack ───────────────────────────────────────────────────────
# transformers 4.47+ introduced a regression on Qwen2.5: post_init() raises
#   TypeError: argument of type 'NoneType' is not iterable
# because ALL_PARALLEL_STYLES is None in that version.
# 4.46.0 is the last confirmed-working release; it is compatible with any
# torch>=2.0 so it works on both cu121 and cu128 runtimes.
transformers==4.46.0

# accelerate provides the device_map="auto" dispatcher.
# 0.33.0 was tested; newer versions are also compatible with transformers 4.46.
accelerate>=0.33.0

# huggingface_hub is a transitive dep of transformers
huggingface_hub>=0.24.0

# peft must be present: bitsandbytes registers PEFT hooks at model-load time
peft>=0.13.2


# ── Quantization ─────────────────────────────────────────────────────────────
# bitsandbytes auto-detects the installed CUDA version and loads the matching
# kernel backend, so it works on both CUDA 12.1 and 12.8 without changes.
# The BitsAndBytesConfig API (load_in_4bit, bnb_4bit_quant_type, etc.) has
# been stable since 0.39; 0.41+ is the minimum for reliable 4-bit NF4 support.
bitsandbytes>=0.41.0


# ── Science RAG ──────────────────────────────────────────────────────────────
# sentence-transformers: embedding model BAAI/bge-small-en-v1.5
sentence-transformers>=2.7.0,<4.0.0

# faiss-cpu 1.9+ requires numpy>=2 on several platform builds.
# Pin to 1.8.0 which is fully compatible with numpy 1.x.
faiss-cpu==1.8.0

# datasets 3.x dropped support for numpy<2.  Pin to the last 2.x release.
datasets==2.20.0

# BM25 hybrid reranker used in rag_science.py
rank_bm25==0.2.2


# ── NER model (Entertainment · News · Philosophy RAG) ────────────────────────
# GLiNER API (GLiNER.from_pretrained + predict_entities) is stable in 0.2.x
gliner>=0.2.0,<0.3.0


# ── Web search ───────────────────────────────────────────────────────────────
# duckduckgo-search 6.x renamed its Python module to `ddgs`.
# All RAG modules do `from ddgs import DDGS`, which requires 6.x.
# 7.x changed the search() text API; pin to 6.2.6 for stability.
duckduckgo-search==6.2.6


# ── Maths RAG ────────────────────────────────────────────────────────────────
# sympy 1.12+ is required for Python 3.12 compatibility
sympy>=1.12


# ── Wikipedia RAG (History · Maths fallback) ─────────────────────────────────
wikipedia==1.4.0


# ── Speech mode ──────────────────────────────────────────────────────────────
openai-whisper>=20231117


# ── HTTP fetching (News · Philosophy RAG) ────────────────────────────────────
requests>=2.31.0
